    <script>

        const API_BASE = window.location.protocol === "file:" ? "http://localhost:5000" : "";
        let allBookings = [];

        // ==================================
        // LOAD BOOKINGS
        // ==================================

        async function loadBookings() {

            try {

                const response = await fetch(`${API_BASE}/api/bookings`);

                const data = await response.json();

                const bookings = Array.isArray(data)
                    ? data
                    : Array.isArray(data.bookings)
                        ? data.bookings
                        : [];

                if (!response.ok || (!Array.isArray(data) && !Array.isArray(data.bookings))) {
                    throw new Error(data.message || "Failed to load bookings");
                }

                allBookings = bookings;

                displayBookings(allBookings);

            } catch (error) {

                console.error(error);

                document.getElementById("bookingTable").innerHTML = `
                    <tr>
                        <td colspan="8" class="no-data">
                            Unable to load bookings.
                        </td>
                    </tr>
                `;

            }

        }


        // ==================================
        // DISPLAY BOOKINGS
        // ==================================

        function displayBookings(bookings) {

            const table = document.getElementById("bookingTable");

            table.innerHTML = "";

            if (bookings.length === 0) {

                table.innerHTML = `
                    <tr>
                        <td colspan="8" class="no-data">
                            No bookings found.
                        </td>
                    </tr>
                `;

                return;
            }


            bookings.forEach(booking => {

                const row = document.createElement("tr");

                row.innerHTML = `

                    <td>
                        <strong>${booking.booking_id}</strong>
                    </td>

                    <td>
                        ${booking.customer_name}
                    </td>

                    <td>
                        ${booking.phone}
                    </td>

                    <td>
                        ${booking.pickup_location || "-"}
                    </td>

                    <td>
                        ${booking.vehicle_model || "-"}
                    </td>

                    <td>
                        ${booking.service || "-"}
                    </td>

                    <td>

                        <select
                            class="status-select"
                            onchange="updateStatus(${booking.id}, this.value, this)"
                        >

                            <option value="Confirmed"
                                ${booking.status === "Confirmed" ? "selected" : ""}>
                                Confirmed
                            </option>

                            <option value="Bike Received"
                                ${booking.status === "Bike Received" ? "selected" : ""}>
                                Bike Received
                            </option>

                            <option value="Inspection"
                                ${booking.status === "Inspection" ? "selected" : ""}>
                                Inspection
                            </option>

                            <option value="In Service"
                                ${booking.status === "In Service" ? "selected" : ""}>
                                In Service
                            </option>

                            <option value="Ready"
                                ${booking.status === "Ready" ? "selected" : ""}>
                                Ready
                            </option>

                            <option value="Completed"
                                ${booking.status === "Completed" ? "selected" : ""}>
                                Completed
                            </option>

                            <option value="Delivered"
                                ${booking.status === "Delivered" ? "selected" : ""}>
                                Delivered
                            </option>

                        </select>

                    </td>

                    <td>

                        <button
                            class="view-btn"
                            onclick="viewBooking(${booking.id})"
                        >
                            View
                        </button>

                    </td>

                `;

                table.appendChild(row);

            });

        }


        // ==================================
        // SEARCH
        // ==================================

        document.getElementById("searchInput")
            .addEventListener("input", function() {

                const search = this.value.toLowerCase();

                const filtered = allBookings.filter(booking =>

                    String(booking.booking_id || "")
                        .toLowerCase()
                        .includes(search)

                    ||

                    String(booking.customer_name || "")
                        .toLowerCase()
                        .includes(search)

                    ||

                    String(booking.phone || "")
                        .toLowerCase()
                        .includes(search)

                    ||

                    String(booking.vehicle_model || "")
                        .toLowerCase()
                        .includes(search)

                );

                displayBookings(filtered);

            });


        // ==================================
        // UPDATE STATUS
        // ==================================

        function cleanPhone(phone) {
            const digits = String(phone || "").replace(/\D/g, "");

            if (digits.length === 10) {
                return `91${digits}`;
            }

            return digits;
        }

        function valueOrDash(value) {
            return value ? String(value) : "-";
        }

        async function fetchJobCardDetails(bookingId) {
            try {
                const response = await fetch(`${API_BASE}/api/job-cards/${encodeURIComponent(bookingId)}`);

                if (!response.ok) {
                    return null;
                }

                return await response.json();
            } catch (error) {
                console.error(error);
                return null;
            }
        }

        function buildCustomerStatusMessage(booking, status, jobCard) {
            const lines = [
                "AYYA Workshop - Booking Update",
                "",
                `Dear ${valueOrDash(booking.customer_name)},`,
                `Your bike service status is updated to: ${status}`,
                "",
                `Booking ID: ${valueOrDash(booking.booking_id)}`,
                `Job Card No: ${valueOrDash(jobCard && jobCard.job_card_no)}`,
                `Phone: ${valueOrDash(booking.phone)}`,
                `Vehicle: ${valueOrDash(booking.vehicle_model)}`,
                `Service: ${valueOrDash(booking.service)}`,
                `Pickup Location: ${valueOrDash(booking.pickup_location)}`,
                `Problem/Complaint: ${valueOrDash(booking.problem || (jobCard && jobCard.problem))}`
            ];

            if (jobCard) {
                lines.push(
                    "",
                    "Job Card Details",
                    `Job Card Status: ${valueOrDash(jobCard.status)}`,
                    `Created At: ${valueOrDash(jobCard.created_at)}`
                );
            }

            if (status === "Completed" || status === "Delivered" || status === "Ready") {
                lines.push(
                    "",
                    "Service Completion Details",
                    `Completion Note: ${valueOrDash(jobCard && jobCard.completion_note)}`,
                    `Completed At: ${valueOrDash(jobCard && jobCard.completed_at)}`
                );
            }

            lines.push("", "Thank you for choosing AYYA Workshop.");

            return lines.join("\n");
        }

        function openCustomerWhatsapp(booking, status, jobCard, shareWindow) {
            const phone = cleanPhone(booking.phone);
            const message = encodeURIComponent(buildCustomerStatusMessage(booking, status, jobCard));
            const url = phone
                ? `https://wa.me/${phone}?text=${message}`
                : `https://wa.me/?text=${message}`;

            if (shareWindow && !shareWindow.closed) {
                shareWindow.location.href = url;
                return;
            }

            window.open(url, "_blank");
        }

        async function updateStatus(id, status, selectElement) {

            const booking = allBookings.find(item => item.id === id);
            const oldStatus = booking ? booking.status : "";
            const shareWindow = window.open("", "_blank");

            if (shareWindow) {
                shareWindow.document.write("<p>Preparing WhatsApp message...</p>");
            }

            try {

                const response = await fetch(
                    `${API_BASE}/api/bookings/${id}/status`,
                    {
                        method: "PUT",

                        headers: {
                            "Content-Type": "application/json"
                        },

                        body: JSON.stringify({
                            status: status
                        })
                    }
                );

                const data = await response.json().catch(() => ({}));

                if (!response.ok || data.success === false) {
                    throw new Error(data.message || `Status update failed (${response.status})`);
                }

                if (booking) {
                    booking.status = status;
                }

                const jobCard = booking ? await fetchJobCardDetails(booking.booking_id) : null;

                if (booking) {
                    openCustomerWhatsapp(booking, status, jobCard, shareWindow);
                } else if (shareWindow && !shareWindow.closed) {
                    shareWindow.close();
                }

                alert("Booking status updated successfully. WhatsApp message opened for the customer.");

                loadBookings();

            } catch (error) {

                if (shareWindow && !shareWindow.closed) {
                    shareWindow.close();
                }

                if (selectElement) {
                    selectElement.value = oldStatus || "Confirmed";
                }

                console.error(error);

                alert(
                    "Unable to update booking status.\n\n" +
                    "Please restart backend/server.js and try again.\n" +
                    error.message
                );

            }

        }


        // ==================================
        // VIEW BOOKING
        // ==================================

        function viewBooking(id) {

            const booking = allBookings.find(
                item => item.id === id
            );

            if (!booking) return;

            alert(
                "Booking ID: " + booking.booking_id +
                "\nCustomer: " + booking.customer_name +
                "\nPhone: " + booking.phone +
                "\nPickup: " + (booking.pickup_location || "-") +
                "\nBike: " + (booking.vehicle_model || "-") +
                "\nService: " + (booking.service || "-") +
                "\nProblem: " + (booking.problem || "-") +
                "\nStatus: " + booking.status
            );

        }


        // Load when page opens
        loadBookings();

    </script>
